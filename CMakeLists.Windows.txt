cmake_minimum_required(VERSION 3.10)

project(LlamaCppChatbot)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ============================================================================
# GPU Backend Detection and Configuration
# ============================================================================
# Auto-detect and configure GPU backend support for llama.cpp
# Priority: 1) Explicit CMake flags, 2) GGML_BACKEND env var, 3) Auto-detect

# Allow explicit backend override via CMake flags
# Use CACHE BOOL instead of option() to allow command-line overrides to work properly
set(GGML_CUDA OFF CACHE BOOL "Enable CUDA GPU backend")
set(GGML_VULKAN OFF CACHE BOOL "Enable Vulkan GPU backend")
set(GGML_HIPBLAS OFF CACHE BOOL "Enable ROCm/HIPBLAS GPU backend")
set(GGML_SYCL OFF CACHE BOOL "Enable SYCL/oneAPI GPU backend")

# Check for explicit backend selection via environment variable
if(DEFINED ENV{GGML_BACKEND} AND NOT GGML_CUDA AND NOT GGML_VULKAN AND NOT GGML_HIPBLAS AND NOT GGML_SYCL)
    string(TOLOWER "$ENV{GGML_BACKEND}" backend_env)
    if(backend_env STREQUAL "cuda")
        set(GGML_CUDA ON)
        message(STATUS "GPU Backend: CUDA (from GGML_BACKEND env)")
    elseif(backend_env STREQUAL "vulkan")
        set(GGML_VULKAN ON)
        message(STATUS "GPU Backend: Vulkan (from GGML_BACKEND env)")
    elseif(backend_env MATCHES "^(hip|rocm|hipblas)$")
        set(GGML_HIPBLAS ON)
        message(STATUS "GPU Backend: ROCm/HIPBLAS (from GGML_BACKEND env)")
    elseif(backend_env MATCHES "^(sycl|oneapi)$")
        set(GGML_SYCL ON)
        message(STATUS "GPU Backend: SYCL/oneAPI (from GGML_BACKEND env)")
    endif()
endif()

# Auto-detect if no backend explicitly set
# Auto-detect if no backend explicitly set
if(NOT GGML_CUDA AND NOT GGML_VULKAN AND NOT GGML_HIPBLAS AND NOT GGML_SYCL)
    # Only auto-detect CUDA if we are using MSVC (since nvcc requires cl.exe)
    if(MSVC)
        find_package(CUDAToolkit QUIET)
        if(CUDAToolkit_FOUND)
            set(GGML_CUDA ON)
            message(STATUS "GPU Backend: CUDA (auto-detected, MSVC environment)")
        endif()
    else()
        message(STATUS "Non-MSVC compiler detected — skipping CUDA auto-detect")
    endif()

    # Fallback to Vulkan if available
    if(NOT GGML_CUDA)
        find_package(Vulkan QUIET)
        if(Vulkan_FOUND)
            set(GGML_VULKAN ON)
            message(STATUS "GPU Backend: Vulkan (auto-detected)")
        else()
            if(EXISTS "/opt/rocm")
                set(GGML_HIPBLAS ON)
                message(STATUS "GPU Backend: ROCm/HIPBLAS (auto-detected)")
            else()
                message(STATUS "GPU Backend: None detected - CPU-only build")
            endif()
        endif()
    endif()
endif()

# Pass GPU backend flags to llama.cpp subdirectory
# These options must be set before add_subdirectory so they propagate
# Always use FORCE to ensure the value is set correctly
if(GGML_CUDA)
    set(GGML_CUDA ON CACHE BOOL "Enable CUDA backend" FORCE)
    message(STATUS "========================================")
    message(STATUS "Enabling CUDA support in llama.cpp")
    message(STATUS "========================================")
    # Verify CUDA can be found
    find_package(CUDAToolkit QUIET)
    if(CUDAToolkit_FOUND)
        message(STATUS "CUDA Toolkit found: ${CUDAToolkit_VERSION}")
    else()
        message(WARNING "CUDA Toolkit not found! CUDA backend may not work.")
        message(WARNING "  Install CUDA Toolkit and ensure it's in PATH")
    endif()
endif()
if(GGML_VULKAN)
    set(GGML_VULKAN ON CACHE BOOL "Enable Vulkan backend" FORCE)
    message(STATUS "========================================")
    message(STATUS "Enabling Vulkan support in llama.cpp")
    message(STATUS "========================================")
endif()
if(GGML_HIPBLAS)
    set(GGML_HIPBLAS ON CACHE BOOL "Enable HIPBLAS backend" FORCE)
    message(STATUS "========================================")
    message(STATUS "Enabling ROCm/HIPBLAS support in llama.cpp")
    message(STATUS "========================================")
endif()
if(GGML_SYCL)
    set(GGML_SYCL ON CACHE BOOL "Enable SYCL backend" FORCE)
    message(STATUS "========================================")
    message(STATUS "Enabling SYCL support in llama.cpp")
    message(STATUS "========================================")
endif()

# Final summary
message(STATUS "")
message(STATUS "GPU Backend Configuration Summary:")
message(STATUS "  GGML_CUDA: ${GGML_CUDA}")
message(STATUS "  GGML_VULKAN: ${GGML_VULKAN}")
message(STATUS "  GGML_HIPBLAS: ${GGML_HIPBLAS}")
message(STATUS "  GGML_SYCL: ${GGML_SYCL}")
message(STATUS "")

# Add llama.cpp subdirectory
add_subdirectory(llama.cpp)

# Create the chatbot executable
add_executable(chatbot main.cpp)

# Link against llama library from llama.cpp
target_link_libraries(chatbot PRIVATE llama)

# Include llama.cpp headers
target_include_directories(chatbot PRIVATE 
    ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp
    ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/common
    ${CMAKE_CURRENT_SOURCE_DIR}/llama.cpp/include
)
